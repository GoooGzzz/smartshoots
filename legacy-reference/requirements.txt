# SMART SHOOTS backend dependencies.
# FIX: the project shipped with NO requirements.txt at all, so `pip install`
# had nothing to go on and the app could not be set up on a fresh machine.
#
# Versions below are picked to work on Python 3.10 through 3.14 (Django
# didn't gain official Python 3.14 support until the 5.2 LTS line, and
# Pillow/PyInstaller needed their own recent releases for 3.14 wheels -
# older pins here caused "Getting requirements to build wheel" / missing
# wheel errors on newer Python installs).

Django==5.2.8
djangorestframework==3.16.1
django-cors-headers==4.3.1
django-filter==24.2
django-fsm==2.8.2
whitenoise==6.6.0
Pillow==11.3.0          # required by ImageField (avatar, business logo); needs 11.x+ for Python 3.13/3.14 wheels

# Production-grade WSGI server for Windows.
# `manage.py runserver` is Django's DEV server: single-threaded, reloads on
# file changes, and prints a "do not use in production" warning. The
# desktop app now uses waitress instead, so the bundled app behaves like a
# real local application instead of a fragile dev server.
waitress==3.0.0

# Only needed if you build a standalone .exe with PyInstaller (see
# build_windows.py). Safe to leave installed for normal `pip install -r`.
# Needs 6.16+ for Python 3.14 support.
pyinstaller==6.20.0
